from flask import Flask, render_template, request, redirect, url_for
from datetime import datetime

app = Flask(__name__)

# လုံခြုံရေးအတွက် Rescue Team သော့ချက် (Secret Key)
RESCUE_SECRET_KEY = "rescue123"

# ယာယီဒေတာ စာရင်း
requests_data = [
    {
        "id": 1,
        "name": "ကိုအောင်အောင်",
        "phone": "09-123456789",
        "people_count": 3,
        "location": "ဗဟန်းမြို့နယ်၊ ရန်ကုန်",
        "urgency": "High",
        "message": "ရေကြီးနေလို့ အိမ်ပေါ်ထပ်မှာ ပိတ်မိနေပါတယ်၊ အစားအသောက်နဲ့ ဆေးဝါး အရေးပေါ်လိုနေပါတယ်။",
        "status": "Pending", # Pending, In Progress, Resolved
        "timestamp": "2026-07-25 10:30 AM",
        "comments": [
            {"team": "ကယ်ဆယ်ရေးအဖွဲ့ (၁)", "text": "အဖွဲ့ထွက်ခွာလာပါပြီ၊ ခဏစောင့်ပါ။"}
        ]
    }
]

@app.route('/')
def rescue_dashboard():
    # Summary အတွက် ကိန်းဂဏန်းများ တွက်ချက်ခြင်း
    total_req = len(requests_data)
    high_urgency = sum(1 for r in requests_data if r["urgency"] == "High")
    resolved_count = sum(1 for r in requests_data if r["status"] == "Resolved")
    
    return render_template('rescue_timeline.html', 
                           requests=requests_data, 
                           total_req=total_req, 
                           high_urgency=high_urgency, 
                           resolved_count=resolved_count)

@app.route('/add_request', methods=['POST'])
def add_request():
    new_item = {
        "id": len(requests_data) + 1,
        "name": request.form.get('name'),
        "phone": request.form.get('phone'),
        "people_count": request.form.get('people_count'),
        "location": request.form.get('location'),
        "urgency": request.form.get('urgency'),
        "message": request.form.get('message'),
        "status": "Pending",
        "timestamp": datetime.now().strftime("%Y-%m-%d %I:%M %p"),
        "comments": []
    }
    requests_data.insert(0, new_item)
    return redirect(url_for('rescue_dashboard'))

@app.route('/update_status/<int:req_id>', methods=['POST'])
def update_status(req_id):
    entered_key = request.form.get('rescue_key')
    new_status = request.form.get('status')
    
    if entered_key == RESCUE_SECRET_KEY:
        for item in requests_data:
            if item["id"] == req_id:
                item["status"] = new_status
                break
    return redirect(url_for('rescue_dashboard'))

@app.route('/add_comment/<int:req_id>', methods=['POST'])
def add_comment(req_id):
    entered_key = request.form.get('rescue_key')
    team_name = request.form.get('team_name')
    comment_text = request.form.get('comment_text')
    
    if entered_key == RESCUE_SECRET_KEY:
        for item in requests_data:
            if item["id"] == req_id:
                item["comments"].append({"team": team_name, "text": comment_text})
                break
    return redirect(url_for('rescue_dashboard'))

if __name__ == '__main__':
    app.run(debug=True, port=5000)
